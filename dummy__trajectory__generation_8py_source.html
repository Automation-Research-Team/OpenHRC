116def getMinJerkTraj(dt, tf, x0, xf):
117
118 t = np.arange(0.0, tf, dt)
119
120 p0 = np.squeeze(x0[["px", "py", "pz"]].values)
121 pf = np.squeeze(xf[["px", "py", "pz"]].values)
122
123 s = t/tf
124 s2 = s * s
125 s3 = s2*s
126 s4 = s3*s
127 s5 = s4*s
128
129 pt = np.empty([3, len(s)])
130 vt = np.empty([3, len(s)])
131 at = np.empty([3, len(s)])
132 for i in range(3):
133 pt[i] = p0[i] + (pf[i] - p0[i]) * (6.0 * s5 - 15.0*s4 + 10.0*s3)
134 vt[i] = (pf[i] - p0[i]) * (30.0 * s4 - 60.0*s3 + 30.0*s2)/tf
135 at[i] = (pf[i] - p0[i]) * (120.0 * s3 - 180.0*s2 + 60.0*s)/(tf*tf)
136
137 # t = t[:, np.newaxis]
138 xt = np.concatenate(
139 [
140 t[:, np.newaxis], np.transpose(
141 pt), np.transpose(vt), np.transpose(at)

143 )
